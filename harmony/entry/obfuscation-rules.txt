# 藏币录 · 混淆规则

# ---- 混淆开关（默认全部注释，保持可读与可排障） ----
# -enable-toplevel-obfuscation     # 顶层作用域名称混淆
# -enable-property-obfuscation     # 属性名混淆：本工程大量用 JSON 字段名与 Record 索引，开启前必须回归
# -enable-filename-obfuscation     # 文件名混淆
# -enable-export-obfuscation       # 导出名混淆
# -compact                         # 去空白
# -remove-log                      # 移除 console 输出（本项目用 hilog，影响小）

# ---- 保留项（开启上面开关时必须保留的内容） ----

  holdings

  userCount
  isAdmin
  rev

  createdAt
